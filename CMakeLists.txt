if(CANE_BUILD_GAME)
 file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/Sly1-Release/Sly1-Release.vcxproj" source_lines REGEX "<ClCompile Include=")
 set(game_sources "")
 foreach(line IN LISTS source_lines)
  string(REGEX REPLACE ".*Include=\"([^\"]+)\".*" "\\1" source "${line}")
  string(REPLACE "\\" "/" source "${source}")
  if(NOT source MATCHES "^\\.\\." AND NOT source STREQUAL "glad.c")
   list(APPEND game_sources "Sly1-Release/${source}")
  endif()
 endforeach()
 add_executable(Sly1-Release ${game_sources} ${imgui_root}/imgui_demo.cpp)
 target_link_libraries(Sly1-Release PRIVATE cane_ui Threads::Threads)
 target_compile_definitions(Sly1-Release PRIVATE CANE_PORTABLE_SHADERS _CRT_SECURE_NO_WARNINGS _ENABLE_EXTENDED_ALIGNED_STORAGE)
 target_include_directories(Sly1-Release PRIVATE "${CMAKE_BINARY_DIR}/generated")
 if(NOT MSVC)
  target_compile_options(Sly1-Release PRIVATE -fno-strict-aliasing -Wno-invalid-offsetof)
 endif()
 file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
 file(GLOB shader_inputs CONFIGURE_DEPENDS "Sly1-Release/*.vert" "Sly1-Release/*.frag" "Sly1-Release/*.geom")
 add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/generated/embedded_shaders.h" COMMAND ${CMAKE_COMMAND} -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}/Sly1-Release -DOUTPUT=${CMAKE_BINARY_DIR}/generated/embedded_shaders.h -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed-shaders.cmake DEPENDS ${shader_inputs} cmake/embed-shaders.cmake VERBATIM)
 target_sources(Sly1-Release PRIVATE "${CMAKE_BINARY_DIR}/generated/embedded_shaders.h")
 foreach(header platform/platform.h Sly1-Release/sw.h Sly1-Release/save.h Sly1-Release/freeze.h Sly1-Release/act.h Sly1-Release/wr.h Sly1-Release/shadow.h Sly1-Release/fader.h Sly1-Release/splicemap.h Sly1-Release/gc.h)
  target_precompile_headers(Sly1-Release PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:${CMAKE_CURRENT_SOURCE_DIR}/${header}>")
 endforeach()
 set_source_files_properties(Sly1-Release/alo.cpp PROPERTIES COMPILE_DEFINITIONS ALO_IMPLEMENTATION SKIP_PRECOMPILE_HEADERS ON)
 if(WIN32)
  if(NOT CANE_FFMPEG_ROOT)
   set(CANE_FFMPEG_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/Libraries/ffmpeg/ffmpeg-master-latest-win64-lgpl-shared")
  endif()
  target_include_directories(Sly1-Release PRIVATE "${CANE_FFMPEG_ROOT}/include")
  target_link_directories(Sly1-Release PRIVATE "${CANE_FFMPEG_ROOT}/lib")
  target_link_libraries(Sly1-Release PRIVATE opengl32 winmm avformat avcodec avutil swscale)
  if(MINGW)
   FetchContent_Declare(xaudio2 URL https://www.nuget.org/api/v2/package/Microsoft.XAudio2.Redist/1.2.13 DOWNLOAD_NAME xaudio2.zip URL_HASH SHA256=bc83d03f6b8615742e583317360336905a104bfa4d30a4d05da65fca424840a2)
   FetchContent_MakeAvailable(xaudio2)
   file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/xaudio2/include")
   file(READ "${xaudio2_SOURCE_DIR}/build/native/include/xaudio2Redist.h" xaudio_header)
   string(REPLACE "<AudioSessionTypes.h>" "<audiosessiontypes.h>" xaudio_header "${xaudio_header}")
   file(WRITE "${CMAKE_BINARY_DIR}/xaudio2/include/xaudio2Redist.h" "${xaudio_header}")
   configure_file("${xaudio2_SOURCE_DIR}/build/native/include/xaudio2fxRedist.h" "${CMAKE_BINARY_DIR}/xaudio2/include/xaudio2fxRedist.h" COPYONLY)
   target_include_directories(Sly1-Release BEFORE PRIVATE "${CMAKE_BINARY_DIR}/xaudio2/include")
   target_compile_definitions(Sly1-Release PRIVATE CANE_XAUDIO_REDIST)
   target_link_libraries(Sly1-Release PRIVATE "${xaudio2_SOURCE_DIR}/build/native/release/lib/x64/xaudio2_9redist.lib")
   install(FILES "${xaudio2_SOURCE_DIR}/build/native/release/bin/x64/xaudio2_9redist.dll" DESTINATION bin)
   install(FILES "${xaudio2_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses/xaudio2)
  else()
   target_link_libraries(Sly1-Release PRIVATE xaudio2)
  endif()
 else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale)
  # Ubuntu 22.04 ships SDL 2.0.20; FAudio requires 2.24 or newer.
  # Prefer an explicitly selected/system development package when compatible.
  find_package(SDL2 2.24 CONFIG QUIET)
  if(NOT SDL2_FOUND)
   set(SDL_SHARED ON CACHE BOOL "" FORCE)
   set(SDL_STATIC OFF CACHE BOOL "" FORCE)
   set(SDL_TEST OFF CACHE BOOL "" FORCE)
   set(SDL2_DISABLE_INSTALL OFF CACHE BOOL "" FORCE)
   FetchContent_Declare(sdl2 URL https://codeload.github.com/libsdl-org/SDL/tar.gz/refs/tags/release-2.30.11 URL_HASH SHA256=cc6136dd964854e8846c679703322f3e2a341d27a06a53f8b3f642c26f1b0cfd)
   FetchContent_MakeAvailable(sdl2)
   set(SDL2_INCLUDE_DIRS "${sdl2_SOURCE_DIR}/include;${sdl2_BINARY_DIR}/include")
   set(SDL2_LIBRARIES SDL2::SDL2)
  endif()
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(faudio URL https://codeload.github.com/FNA-XNA/FAudio/tar.gz/refs/tags/25.02 URL_HASH SHA256=eab667d231e5036b0e5c3c6f89fb4ea792284493d46161bc056a88a6dadc2683)
  FetchContent_MakeAvailable(faudio)
  if(TARGET FAudio-shared)
   set(cane_audio_target FAudio-shared)
  else()
   set(cane_audio_target FAudio-static)
  endif()
  target_link_libraries(Sly1-Release PRIVATE PkgConfig::FFMPEG ${cane_audio_target})
 endif()
endif()
